10 Logical Programs with Output

1. Even or Odd

Check whether a number is even or odd.

let num1 = 25;
if (num1 % 2 === 0) {
console.log(num1 + " is an Even Number");
} else {
console.log(num1 + " is an Odd Number");
}
OUTPUT :
25 is an Odd Number

2. Largest Number

Find the largest among three numbers.

let a = 25;
let b = 45;
let c = 35;
let largest = Math.max(a, b, c);
console.log("Numbers: " + a + ", " + b + ", " + c);
console.log("Largest = " + largest);
OUTPUT :
Numbers: 25, 45, 35
Largest = 45

3. Prime Number

Check whether a number is prime.

let num3 = 17;
let isPrime = true;
if (num3 < 2) {
isPrime = false;
}
for (let i = 2; i <= Math.sqrt(num3); i++) {
if (num3 % i === 0) {
isPrime = false;
break;
}
}
if (isPrime) {
console.log(num3 + " is a Prime Number");
} else {
console.log(num3 + " is Not a Prime Number");
}
OUTPUT :
17 is a Prime Number

4. Factorial

Calculate the factorial of a number.

let num4 = 5;
let factorial = 1;
for (let i = 1; i <= num4; i++) {
factorial = factorial * i;
}
console.log("Factorial of " + num4 + " = " + factorial);
OUTPUT :
Factorial of 5 = 120

5. Fibonacci Series

Generate the Fibonacci series.

let n = 8;
let first = 0;
let second = 1;
let fibonacci = [];
for (let i = 0; i < n; i++) {
fibonacci.push(first);
let next = first + second;
first = second;
second = next;
}
console.log("Fibonacci Series: " + fibonacci.join(" , "));
OUTPUT :
Fibonacci Series: 0 , 1 , 1 , 2 , 3 , 5 , 8 , 13

6. Palindrome

Check whether a number is a palindrome.

let num6 = 121;
let original = num6;
let reversed6 = 0;
while (num6 > 0) {
let digit = num6 % 10;
reversed6 = reversed6 * 10 + digit;
num6 = Math.floor(num6 / 10);
}
if (original === reversed6) {
console.log(original + " is a Palindrome");
} else {
console.log(original + " is Not a Palindrome");
}
OUTPUT :
121 is a Palindrome

7. Reverse Number

Reverse a given number.

let num7 = 12345;
let temp = num7;
let reversed7 = 0;
while (temp > 0) {
let digit = temp % 10;
reversed7 = reversed7 * 10 + digit;
temp = Math.floor(temp / 10);
}
console.log("Reverse of " + num7 + " = " + reversed7);
OUTPUT :
Reverse of 12345 = 54321

8. Sum of Digits

Calculate the sum of digits.

let num8 = 12345;
let temp8 = num8;
let sum = 0;
while (temp8 > 0) {
sum = sum + (temp8 % 10);
temp8 = Math.floor(temp8 / 10);
}
console.log("Sum of digits of " + num8 + " = " + sum);
OUTPUT :
Sum of digits of 12345 = 15

9. Armstrong Number

Check whether a number is an Armstrong number.

let num9 = 153;
let original9 = num9;
let digits = String(num9).length;
let temp9 = num9;
let armstrongSum = 0;
while (temp9 > 0) {
let digit = temp9 % 10;
armstrongSum = armstrongSum + Math.pow(digit, digits);
temp9 = Math.floor(temp9 / 10);
}
if (armstrongSum === original9) {
console.log(original9 + " is an Armstrong Number");
} else {
console.log(original9 + " is Not an Armstrong Number");
}
OUTPUT :
153 is an Armstrong Number

10. Leap Year

Check whether a year is a leap year.

let year = 2024;
let leap = (year % 400 === 0) ||
(year % 4 === 0 && year % 100 !== 0);
if (leap) {
console.log(year + " is a Leap Year");
} else {
console.log(year + " is Not a Leap Year");
}
OUTPUT :
2024 is a Leap Year
